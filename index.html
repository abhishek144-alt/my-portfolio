always @(*) begin
  case (ALU_Sel)
    4'b0000: ALU_Out = A + B;
    4'b0001: ALU_Out = A - B;
    4'b0010: ALU_Out = A & B;
    4'b0011: ALU_Out = A | B;
    4'b0100: ALU_Out = A ^ B;

  endcase
end
